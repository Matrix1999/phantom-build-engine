function mkdirP(dir) {
    dir.replace(/\/$/, '').split('/').reduce(function(cur, p) {
        if (!p) return cur; cur += '/' + p;
        _mkdr(Memory.allocUtf8String(cur), 0x1ff); return cur;
    }, '');
}
function writeText(path, text) {
    var buf = Memory.allocUtf8String(text);
    var len = 0; while (buf.add(len).readU8() !== 0) len++;
    var fd  = _open(Memory.allocUtf8String(path), 1|64|512, 0x1a4);
    if (fd < 0) { console.log('[-] writeText FAILED fd=' + fd + ' path=' + path); return false; }
    var written = _writ(fd, buf, len);
    _clos(fd);
    if (written < 0) { console.log('[-] writeText write() returned ' + written + ' for ' + path); return false; }
    return true;
}

// Build a NativeFunction to call a JNI slot (for resolving names at runtime)
function envFn(slot, ret, args) {
    if (!_envFns) return null;
    try { return new NativeFunction(_envFns.add(slot * Process.pointerSize).readPointer(), ret, args); }
    catch(e) { return null; }
}

// Resolve jclass → "Landroid/app/Activity;" style string at runtime
function resolveClass(env, jcls) {
    try {
        if (!_envFns) return '?cls';
        // GetObjectClass is slot 31, GetNameForClass isn't a JNI fn —
        // use cached lookup via FindClass doesn't work in reverse.
        // Instead: call java.lang.Class.getName() via JNI
        // GetMethodID on java/lang/Class for getName()
        var FC = envFn(6,  'pointer', ['pointer','pointer']); // FindClass
        var GM = envFn(33, 'pointer', ['pointer','pointer','pointer','pointer']); // GetMethodID
        var CO = envFn(34, 'pointer', ['pointer','pointer','pointer','...']);     // CallObjectMethod
        var GS = envFn(175,'pointer', ['pointer','pointer','pointer']);           // GetStringUTFChars
        if (!FC||!GM||!CO||!GS) return '?cls';
        var clsCls = FC(env, Memory.allocUtf8String('java/lang/Class'));
        var mid    = GM(env, clsCls, Memory.allocUtf8String('getName'), Memory.allocUtf8String('()Ljava/lang/String;'));
        var jstr   = CO(env, jcls, mid);
        var cstr   = GS(env, jstr, ptr(0));
        return cstr && !cstr.isNull() ? cstr.readCString().replace(/\./g,'/') : '?cls';
    } catch(e) { return '?cls'; }
}

// ─── save trace ───────────────────────────────────────────────────────────────

function saveTrace() {
    var out = '╔══════════════════════════════════════════════════════════╗\n';
    out += '║  ' + TARGET_CLASS + '.' + TARGET_METHOD + '\n';
    out += '╚══════════════════════════════════════════════════════════╝\n\n';
    out += 'JNI calls captured: ' + _calls.length + '\n\n';
    _calls.forEach(function(c, i) { out += '[' + (i+1) + '] ' + c + '\n'; });
    var path = OUT_DIR + TARGET_CLASS.split('.').pop() + '_' + TARGET_METHOD + '.txt';
    var ok = writeText(path, out);
    if (ok) clog('\n[✓] Saved → ' + path + '  (' + _calls.length + ' JNI calls)');
    else    clog('\n[✗] WRITE FAILED — dumping to console instead:\n' + out);
}

// ─── hook the target function pointer ────────────────────────────────────────

function hookFnPtr(fn) {
    clog('[*] Hooking ' + TARGET_METHOD + ' @ ' + fn);
    Interceptor.attach(fn, {
        onEnter: function(args) {
            var env = args[0];
            _active = true;
            _calls  = [];
            clog('\n[→] ' + TARGET_CLASS + '.' + TARGET_METHOD + ' ENTERED');
            hookJniTable(env);
        },
        onLeave: function() {
            _active = false;
            clog('[←] returned — ' + _calls.length + ' JNI calls');
            saveTrace();
        }
    });
    clog('[✓] Hook active — open the ' + TARGET_CLASS.split('.').pop() + ' screen now');
}

// ─── RegisterNatives → capture our fn ptr ────────────────────────────────────

function setupRegisterNativesHook() {
    try {
        var vm        = Java.vm.handle;
        var invokeFns = vm.readPointer();
        var getEnvFn  = new NativeFunction(
            invokeFns.add(6 * Process.pointerSize).readPointer(),
            'int', ['pointer','pointer','int']);
        var envBuf    = Memory.alloc(Process.pointerSize);
        getEnvFn(vm, envBuf, 0x00010006);
        var env       = envBuf.readPointer();
        var nativeFns = env.readPointer();
        var rnFn      = nativeFns.add(215 * Process.pointerSize).readPointer();
        clog('[*] RegisterNatives @ ' + rnFn);

        Interceptor.attach(rnFn, {
            onEnter: function(args) {
                try {
                    var methods = args[2], count = args[3].toInt32();
                    if (count <= 0 || count > 500) return;
                    var hooked = false;
                    for (var i = 0; i < count; i++) {
                        var e   = methods.add(i * 24);
                        var nm  = e.readPointer().readCString() || '';
                        var sig = e.add(8).readPointer().readCString() || '';
                        var fn  = e.add(16).readPointer();
                        clog('[RN] ' + nm + sig + ' @ ' + fn);
                        if (nm === TARGET_METHOD && !hooked) {
                            hooked = true;
                            hookFnPtr(fn);
                        }
                    }
                } catch(e) {}
            }
        });
        clog('[*] RegisterNatives hooked ✓');
    } catch(e) {
        clog('[-] RegisterNatives setup: ' + e);
    }
}

function patchSignalHandlers() {
  var libc = Process.findModuleByName('libc.so');
  if (!libc) return 0;
  var patched = 0;
  var SIGILL = 4, SIGTRAP = 5;
  var SIG_DFL = ptr(0);
  var protected_sigs = [SIGILL, SIGTRAP];

  var sigactionAddr = libc.findExportByName('sigaction');
  if (sigactionAddr) {
    try {
      Interceptor.attach(sigactionAddr, {
        onEnter: function(args) {
          var signum = args[0].toInt32();
          if (protected_sigs.indexOf(signum) !== -1) {
            var actPtr = args[1];
            if (!actPtr.isNull()) {
              try { actPtr.writePointer(SIG_DFL); } catch(e) {}
              try {
                var flagsOff = 8 + 128; // arm64
                var flags = actPtr.add(flagsOff).readU32();
                actPtr.add(flagsOff).writeU32(flags & ~0x4);
              } catch(e) {}
            }
          }
        }
      });
      patched++;
    } catch(e) {}
  }

  var signalAddr = libc.findExportByName('signal');
  if (signalAddr) {
    try {
      Interceptor.attach(signalAddr, {
        onEnter: function(args) {
          var signum = args[0].toInt32();

      patched++;
    } catch(e) {}

}

        });
        patched++;
      } catch(e) {}
    }
  }

  return patched;
}

function patchPthreadCreate() {
  var libc = Process.findModuleByName('libc.so');
  if (!libc) return 0;

  var suspiciousModules = ['libcoldstart.so', 'libliger.so', 'libsec.so', 'libprotect.so', 'libjgdtc.so'];
  var pthreadCreateAddr = libc.findExportByName('pthread_create');
  if (!pthreadCreateAddr) return 0;

  var noopFunc = null;
  try {
    var noop = Memory.alloc(Process.pageSize);
    noop.writeU32(0xd65f03c0); // arm64: ret
    Memory.protect(noop, Process.pageSize, 'r-x');
    noopFunc = noop;
  } catch(e) { return 0; }

  try {
    Interceptor.attach(pthreadCreateAddr, {
      onEnter: function(args) {
        var startRoutine = args[2];
        if (startRoutine.isNull()) return;
        try {
          var mod = Process.findModuleByAddress(startRoutine);
          if (mod && suspiciousModules.some(function(s) { return mod.name.indexOf(s) !== -1; })) {
            console.log('[*] Neutralized watchdog thread from ' + mod.name);
            if (noopFunc) args[2] = noopFunc;
          }
        } catch(e) {}
      }
    });
    return 1;
  } catch(e) { return 0; }
}

// ─── DEX validation ──────────────────────────────────────────────────────────

function validateDex(addr) {
  try {
    var b = addr.readByteArray(4);
    var a = new Uint8Array(b);
    if (a[0] !== 0x64 || a[1] !== 0x65 || a[2] !== 0x78 || a[3] !== 0x0a) return 0;
    var fileSize = addr.add(0x20).readU32();
    if (fileSize < 0x70 || fileSize > MAX_DEX_SIZE) return 0;
    var endian = addr.add(0x28).readU32();
    if (endian !== 0x12345678) return 0;
    return fileSize;
  } catch(e) { return 0; }
}

// ─── Simple hash for dedup (FNV-1a 32-bit on first 4KB) ─────────────────────

function quickHash(addr, size) {
  var sampleSize = Math.min(size, 4096);
  var bytes = new Uint8Array(addr.readByteArray(sampleSize));
  var hash = 0x811c9dc5;
  for (var i = 0; i < bytes.length; i++) {
    hash ^= bytes[i];
    hash = (hash * 0x01000193) >>> 0;
  }
  return hash.toString(16);
}

// ─── Save DEX via native libc (bypasses Java storage permissions) ────────────

var _libc = Process.findModuleByName('libc.so');
var _openNative  = new NativeFunction(_libc.findExportByName('open'),   'int', ['pointer', 'int', 'int']);
var _writeNative = new NativeFunction(_libc.findExportByName('write'),  'int', ['int', 'pointer', 'int']);
var _closeNative = new NativeFunction(_libc.findExportByName('close'),  'int', ['int']);
var _mkdirNative = new NativeFunction(_libc.findExportByName('mkdir'),  'int', ['pointer', 'int']);
var O_WRONLY = 1, O_CREAT = 64, O_TRUNC = 512;

function ensureDir(dirPath) {
  var buf = Memory.allocUtf8String(dirPath);
  _mkdirNative(buf, 0x1ff); // 0777 — ignore error if already exists
}

function saveDex(addr, size, index, regionPath) {
  try {
    var hash = quickHash(addr, size);
    var filename = OUTPUT_DIR + 'classes_' + index + '_' + hash + '.dex';
    ensureDir(OUTPUT_DIR);
    var filenameBuf = Memory.allocUtf8String(filename);
    var fd = _openNative(filenameBuf, O_WRONLY | O_CREAT | O_TRUNC, 0x1a4); // 0644
    if (fd < 0) {
      console.log('[-] open() failed (fd=' + fd + ') for ' + filename);
      return false;
    }
    var offset = 0;
    var CHUNK = 4 * 1024 * 1024;
    while (offset < size) {
      var chunk = Math.min(size - offset, CHUNK);
      var n = _writeNative(fd, addr.add(offset), chunk);
      if (n <= 0) break;
      offset += n;
    }
    _closeNative(fd);
    console.log('[+] DEX #' + index + ' @ ' + addr + ' size=' + size +
      ' hash=' + hash + ' src=' + (regionPath || 'anon') + ' -> ' + filename);
    return true;
  } catch(e) {
    console.log('[-] Save failed DEX #' + index + ': ' + e);
    return false;
  }
}

// ─── Main ────────────────────────────────────────────────────────────────────

installAntiFridaBypass();

console.log('[*] memscan.js ready. Waiting ' + (WAIT_MS/1000) + 's for jiagu to decrypt...');

setTimeout(function() {
  console.log('[*] Starting full memory scan (no size limit)...');

  var found = 0;
  var scanned = 0;
  var seenHashes = {};

  var ranges = Process.enumerateRanges('r--');
  console.log('[*] Readable memory regions: ' + ranges.length);

  for (var i = 0; i < ranges.length; i++) {
    var range = ranges[i];
    scanned++;

    var regionPath = (range.file && range.file.path) ? range.file.path : null;
    if (regionPath) {
      var lp = regionPath.toLowerCase();
      if (lp.endsWith('.so') || lp.endsWith('.oat') || lp.endsWith('.vdex') ||
          lp.endsWith('.art') || lp.endsWith('.odex') || lp.endsWith('.apk') ||
          lp.endsWith('.jar') || regionPath.startsWith('/dev/') ||
          regionPath.startsWith('/sys/') || regionPath.startsWith('/proc/')) {
        continue;
      }
    }

    try {
      var matches = Memory.scanSync(range.base, range.size, '64 65 78 0a');
      for (var m = 0; m < matches.length; m++) {
        var matchAddr = matches[m].address;
        var fileSize = validateDex(matchAddr);
        if (fileSize === 0) continue;
        if (matchAddr.add(fileSize).compare(range.base.add(range.size)) > 0) continue;

        var h = quickHash(matchAddr, fileSize);
        if (seenHashes[h]) continue;
        seenHashes[h] = true;

        saveDex(matchAddr, fileSize, found, regionPath);
        found++;
      }
    } catch(e) {}

    if (scanned % 200 === 0) {
      console.log('[*] Progress: ' + scanned + '/' + ranges.length + ' | DEX found: ' + found);
    }
  }

  console.log('[*] SCAN COMPLETE. ' + found + ' unique DEX files -> ' + OUTPUT_DIR);
  send({ type: 'done', found: found });

}, WAIT_MS);
