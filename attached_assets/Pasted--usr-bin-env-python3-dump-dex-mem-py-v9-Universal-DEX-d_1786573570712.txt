#!/usr/bin/env python3
"""
dump_dex_mem.py  v9  —  Universal DEX dumper
=============================================
Handles ALL major Android protectors automatically:

  dpt-shell  → MODE A: static offline restore (no app running needed)
                MODE B: memory scan (optional, for XOR key recovery)

  ijiami / bangcle / general apps
             → MODE B: /proc/PID/mem scan (app must be open)
               Scans ashmem, memfd, dalvik, anon regions — catches all
               runtime-decrypted DEX without Frida.

Auto-detects protection type and picks the right strategy.
Filters out .jar / system regions — only real app DEX saved.
Auto-collects, renames classes_N.dex and zips to /sdcard/dump_output.zip.

─────────────────────────────────────────────────────────────────────────
USAGE (same command for every app):

  # No PID — static only (works for dpt-shell without opening the app):
  su -c "/data/data/com.termux/files/usr/bin/python3 /sdcard/dump_dex_mem.py 0 /data/local/tmp/dpt_out com.example.app"

  # With PID — static + memory scan (needed for ijiami / general apps):
  PID=$(su -c "pidof com.example.app") && echo $PID
  su -c "/data/data/com.termux/files/usr/bin/python3 /sdcard/dump_dex_mem.py $PID /data/local/tmp/dpt_out com.example.app"

STEPS:
  1. su -c "rm -rf /data/local/tmp/dpt_out"
  2. su -c "mkdir -p /data/local/tmp/dpt_out && chmod 777 /data/local/tmp/dpt_out"
  3. (for ijiami) open the app first and wait for it to fully load
  4. PID=$(su -c "pidof com.example.app") && echo $PID   [skip for dpt-shell]
  5. run the command above
  6. find /sdcard/dump_output.zip  — ready to decompile with jadx
─────────────────────────────────────────────────────────────────────────
"""

import sys, os, struct, time, zipfile

# ── CLI ───────────────────────────────────────────────────────────────────
if len(sys.argv) < 2:
    print("Usage: dump_dex_mem.py <PID|0> [OUT_DIR] [PACKAGE_OR_APK]")
    sys.exit(1)

PID     = int(sys.argv[1])
OUT_DIR = sys.argv[2] if len(sys.argv) >= 3 else '/data/local/tmp/dpt_out'
TARGET  = sys.argv[3] if len(sys.argv) >= 4 else None

MAX_SZ  = 80 * 1024 * 1024
MIN_SZ  = 112
CHUNK   = 4 * 1024 * 1024

os.makedirs(OUT_DIR, exist_ok=True)
maps_path = f'/proc/{PID}/maps'
mem_path  = f'/proc/{PID}/mem'
DEX_MAGIC = b'dex\n'

seen_sigs = {}
mem_count = 0


# ═══════════════════════════════════════════════════════════════════════════
# DEX helpers
# ═══════════════════════════════════════════════════════════════════════════

def read_uleb128(data, pos):
    result, shift = 0, 0
    while True:
        b = data[pos]; pos += 1
        result |= (b & 0x7f) << shift
        if not (b & 0x80): break
        shift += 7
    return result, pos


def parse_dex_methods(dex):
    """Return dict: method_idx -> (code_off, insns_off, insns_size_bytes)"""
    out = {}
    try:
        class_defs_size = struct.unpack_from('<I', dex, 96)[0]
        class_defs_off  = struct.unpack_from('<I', dex, 100)[0]
    except Exception:
        return out
    for ci in range(class_defs_size):
        cd = class_defs_off + ci * 32
        cdo = struct.unpack_from('<I', dex, cd + 24)[0]
        if not cdo: continue
        try:
            pos = cdo
            sf, pos = read_uleb128(dex, pos)
            inf, pos = read_uleb128(dex, pos)
            dm, pos  = read_uleb128(dex, pos)
            vm, pos  = read_uleb128(dex, pos)
            for _ in range(sf + inf):
                _, pos = read_uleb128(dex, pos)
                _, pos = read_uleb128(dex, pos)
            midx = 0
            for _ in range(dm + vm):
                d,  pos = read_uleb128(dex, pos)
                _,  pos = read_uleb128(dex, pos)
                co, pos = read_uleb128(dex, pos)
                midx += d
                if not co: continue
                isz = struct.unpack_from('<I', dex, co + 12)[0]
                out[midx] = (co, co + 16, isz * 2)
        except Exception:
            continue
    return out


def is_empty(data):
    """True if bytes look like dpt NOP placeholder."""
    s = set(data[:min(8, len(data))])
    return s <= {0x00, 0x0e, 0x01}


def xor_decrypt(enc, key):
    if key == 0:
        return bytes(enc)
    kb = struct.pack('<I', key & 0xFFFFFFFF)
    return bytes(b ^ kb[i & 3] for i, b in enumerate(enc))


# ═══════════════════════════════════════════════════════════════════════════
# OoooooOooo parser  (dpt-shell only)
# ═══════════════════════════════════════════════════════════════════════════

def parse_multicode(raw):
    """
    dpt MultiDexCode binary (OoooooOooo or any renamed asset):
      uint16 version
      uint16 dex_count
      uint32[dex_count] offsets
      per dex @ offset:
        uint16 method_count
        per method: uint32 method_idx, uint32 insns_size, byte[insns_size]
    Returns list of dicts {method_idx: encrypted_bytes}
    """
    if len(raw) < 8: return []
    version   = struct.unpack_from('<H', raw, 0)[0]
    dex_count = struct.unpack_from('<H', raw, 2)[0]
    if dex_count == 0 or dex_count > 200: return []
    if version > 20: return []

    offsets = [struct.unpack_from('<I', raw, 4 + i*4)[0] for i in range(dex_count)]

    result = []
    for off in offsets:
        methods = {}
        if off + 2 > len(raw):
            result.append(methods); continue
        mc = struct.unpack_from('<H', raw, off)[0]
        pos = off + 2
        for _ in range(mc):
            if pos + 8 > len(raw): break
            midx = struct.unpack_from('<I', raw, pos)[0]
            isz  = struct.unpack_from('<I', raw, pos+4)[0]
            pos += 8
            if 0 < isz <= MAX_SZ and pos + isz <= len(raw):
                methods[midx] = raw[pos:pos+isz]
            pos += isz
        result.append(methods)

    total = sum(len(d) for d in result)
    print(f'[*] MultiDexCode v{version}: {len(result)} dex, {total} methods total')
    return result


def looks_like_multicode(raw):
    """Heuristic: is this raw bytes a valid MultiDexCode binary?"""
    if len(raw) < 12: return False
    version   = struct.unpack_from('<H', raw, 0)[0]
    dex_count = struct.unpack_from('<H', raw, 2)[0]
    if version > 20 or dex_count == 0 or dex_count > 200: return False
    off0 = struct.unpack_from('<I', raw, 4)[0]
    if off0 >= len(raw): return False
    mc = struct.unpack_from('<H', raw, off0)[0]
    return mc > 0

def find_apk(target):
    """Resolve package name or APK path to an actual .apk file path."""
    if target and os.path.isfile(target):
        return target

    pkg = target
    if pkg:
        try:
            pm_out = os.popen(f'pm path {pkg} 2>/dev/null').read().strip()
            if pm_out.startswith('package:'):
                p = pm_out[8:].strip()
                if os.path.isfile(p):
                    print(f'[*] APK found via pm: {p}')
                    return p
        except Exception:
            pass

    try:
        for entry in os.listdir('/data/app'):
            if pkg and pkg not in entry:
                continue
            for name in ('base.apk',):
                p = os.path.join('/data/app', entry, name)
                if os.path.isfile(p):
                    return p
                for sub in os.listdir(os.path.join('/data/app', entry)):
                    p2 = os.path.join('/data/app', entry, sub, name)
                    if os.path.isfile(p2):
                        return p2
    except Exception:
        pass

    if PID > 0:
        try:
            for line in open(maps_path):
                parts = line.split()
                if len(parts) >= 6 and parts[5].endswith('.apk') and '/data/app/' in parts[5]:
                    return parts[5]
        except Exception:
            pass

    return None


def extract_from_apk(apk_path):
    """
    Returns:
      dex_files  : list of (entry_name, bytes)
      code_asset : bytes or None  (MultiDexCode binary — dpt only)
      asset_name : str
      is_dpt     : bool
    """
    dex_files  = []
    code_asset = None
    asset_name = '(not found)'

    try:
        with zipfile.ZipFile(apk_path, 'r') as z:
            names = z.namelist()

            for name in sorted(names):
                if name.startswith('classes') and name.endswith('.dex'):
                    dex_files.append((name, z.read(name)))

            for name in names:
                if not name.startswith('assets/'):
                    continue
                low = name.lower()
                if any(x in low for x in ('.so', 'baseline', 'protected', 'dexopt', 'kotlin')):
                    continue
                raw = z.read(name)
                if looks_like_multicode(raw):
                    code_asset = raw
                    asset_name = name
                    break

    except Exception as e:
        print(f'[!] APK read error: {e}')

    is_dpt = code_asset is not None
    return dex_files, code_asset, asset_name, is_dpt


def detect_protection(apk_path):
    """
    Quick scan of APK to detect protector type.
    Returns: 'dpt-shell' | 'ijiami' | 'general'
    """
    if not apk_path:
        return 'general'
    try:
        with zipfile.ZipFile(apk_path, 'r') as z:
            names = z.namelist()
            names_lower = [n.lower() for n in names]

            # dpt-shell: has OoooooOooo-style asset + dpt native lib
            for name in names:
                raw = z.read(name) if name.startswith('assets/') else b''
                if raw and looks_like_multicode(raw):
                    return 'dpt-shell'

            # ijiami: has ijm_libs or libexec-main or similar
            for n in names_lower:
                if any(x in n for x in ('ijm_', 'ijiami', 'libexec', 'libDexHelper'.lower(),
                                         'libprotect', 'libbangcle')):
                    return 'ijiami'

    except Exception:
        pass
    return 'general'


# ═══════════════════════════════════════════════════════════════════════════
# XOR key recovery  (dpt-shell only)
# ═══════════════════════════════════════════════════════════════════════════

def recover_xor_key(dex, dex_methods, oooo_methods):
    votes = {}
    for midx, (co, io, isz) in dex_methods.items():
        if isz < 4 or midx not in oooo_methods: continue
        mem_b = dex[io:io+isz]
        enc   = oooo_methods[midx]
        if len(enc) < isz or is_empty(mem_b): continue
        key = struct.unpack_from('<I', bytes(mem_b[i] ^ enc[i] for i in range(4)))[0]
        dec = xor_decrypt(enc[:isz], key)
        if dec == mem_b[:isz]:
            votes[key] = votes.get(key, 0) + 1
    if not votes: return None
    best = max(votes, key=lambda k: votes[k])
    print(f'[*] XOR key recovered: 0x{best:08x}  (from {votes[best]} method(s))')
    return best


def try_xor_key_zero_validate(dex_methods, oooo_methods):
    VALID_OPCODES = set(range(0x00, 0x3e)) | {
        0x44,0x45,0x46,0x47,0x48,0x49,0x4a,0x4b,0x4c,0x4d,0x4e,0x4f,
        0x52,0x54,0x56,0x58,0x5a,0x5c,0x5e,0x60,0x62,0x6e,0x6f,0x70,
        0x71,0x72,0x74,0x76,0x78,0x7a,0x7c,0x7e,0x80,0x82,0x84,0x86,
        0x88,0x8a,0x8c,0x8e,0x90,0x92,0x94,0x96,0x98,0x9a,0x9c,0x9e,
        0xa0,0xa2,0xa4,0xa6,0xa8,0xaa,0xac,0xae,0xb0,0xb2,0xb4,0xb6,
        0xb8,0xba,0xbb,0xbc,0xbd,0xbe,0xbf,0xc0,0xc2,0xc4,0xc6,0xc8,
        0xca,0xd8,0xda,0xdc,0xde,0xe0,0xe2,0xe4,0xe6,0xe8,0xea,0xec,
        0xee,0xf0,0xf2,0xf4,0xf6,0xf8,0xfa,0xfc,0xfe}
    good, bad = 0, 0
    for midx, enc in list(oooo_methods.items())[:200]:
        if not enc: continue
        if enc[0] in VALID_OPCODES:
            good += 1
        else:
            bad += 1
    if good + bad == 0: return False
    ratio = good / (good + bad)
    print(f'[*] XOR key=0 validation: {good} valid, {bad} invalid  ({ratio*100:.0f}%)')
    return ratio >= 0.60


def match_dex_to_oooo(dex_methods, oooo_list):
    best_i, best_n = 0, 0
    dm_keys = set(dex_methods.keys())
    for i, od in enumerate(oooo_list):
        n = len(dm_keys & set(od.keys()))
        if n > best_n:
            best_n = n; best_i = i
    return best_i, best_n


def patch_dex(dex, dex_methods, oooo_methods, xor_key):
    buf = bytearray(dex)
    patched = skipped = 0
    for midx, enc in oooo_methods.items():
        if midx not in dex_methods: skipped += 1; continue
        co, io, isz = dex_methods[midx]
        if len(enc) != isz: skipped += 1; continue
        if not is_empty(bytes(buf[io:io+isz])): continue
        buf[io:io+isz] = xor_decrypt(enc, xor_key)
        patched += 1
    return bytes(buf), patched, skipped

# MODE A — Static APK restore  (dpt-shell, no app running needed)
# ═══════════════════════════════════════════════════════════════════════════

def run_static_mode():
    print('\n[*] ══════════════════════════════════════')
    print('[*] MODE A — Static APK restore')
    print('[*] ══════════════════════════════════════')

    apk = find_apk(TARGET)
    if not apk:
        print('[!] Could not find APK. Pass path as 3rd arg:')
        print(f'    python3 dump_dex_mem.py 0 {OUT_DIR} /sdcard/target.apk')
        return []

    print(f'[*] APK: {apk}')
    dex_files, code_asset, asset_name, is_dpt = extract_from_apk(apk)

    if not dex_files:
        print('[!] No classes.dex found in APK.')
        return []

    print(f'[*] Found {len(dex_files)} DEX file(s): {[n for n,_ in dex_files]}')

    # ── Not dpt-shell: save static DEX as-is ──────────────────────────────
    if code_asset is None:
        print('[!] No OoooooOooo asset found → not dpt-shell.')
        print('    Saving DEX files as-is (static dump).')
        saved = []
        for name, data in dex_files:
            safe = name.replace('/', '_')
            out  = os.path.join(OUT_DIR, f'static_{safe}')
            open(out, 'wb').write(data)
            print(f'[+] {out}  ({len(data):,} bytes)')
            saved.append(out)
        return saved

    # ── dpt-shell: patch hollow methods with OoooooOooo ───────────────────
    print(f'[*] OoooooOooo asset: {asset_name}  ({len(code_asset):,} bytes)')
    oooo_list = parse_multicode(code_asset)
    if not oooo_list:
        print('[!] Failed to parse MultiDexCode binary.')
        return []

    restored_files = []
    for dex_name, hollow_dex in dex_files:
        print(f'\n[*] Processing {dex_name} ({len(hollow_dex):,} bytes) ...')
        dex_methods = parse_dex_methods(hollow_dex)
        if not dex_methods:
            print('    [!] No parseable methods — saving as-is.')
            out = os.path.join(OUT_DIR, f'static_{dex_name.replace("/","_")}')
            open(out, 'wb').write(hollow_dex)
            restored_files.append(out); continue

        print(f'    Methods with code items: {len(dex_methods)}')
        oooo_idx, overlap = match_dex_to_oooo(dex_methods, oooo_list)
        if overlap == 0:
            print('    [!] No overlap with OoooooOooo — saving as-is.')
            out = os.path.join(OUT_DIR, f'static_{dex_name.replace("/","_")}')
            open(out, 'wb').write(hollow_dex)
            restored_files.append(out); continue

        oooo_methods = oooo_list[oooo_idx]
        print(f'    Matched dex[{oooo_idx}] overlap={overlap}, '
              f'total={len(oooo_methods)} methods')

        if try_xor_key_zero_validate(dex_methods, oooo_methods):
            xor_key = 0
            print('    XOR key = 0 (no encryption)')
        else:
            print('    key=0 failed — patching anyway (provide PID to crack key)')
            xor_key = 0

        patched_dex, n_pat, n_skip = patch_dex(
            hollow_dex, dex_methods, oooo_methods, xor_key)

        out = os.path.join(OUT_DIR, f'restored_{dex_name.replace("/","_")}')
        open(out, 'wb').write(patched_dex)
        print(f'    [+] {out}')
        print(f'        patched={n_pat}  skipped={n_skip}')
        restored_files.append(out)

    return restored_files

# MODE B — Memory scan (ijiami / bangcle / general / dpt fallback)
# Covers: anonymous, ashmem, memfd, dalvik, /data/app, /data/data, /data/user
# Blocks: .jar, system .so, .oat/.vdex/.art, /system, /vendor, /apex
# ═══════════════════════════════════════════════════════════════════════════

def should_scan(pathname, perm):
    """
    Unified region filter for all protectors:
    - ijiami   → /dev/ashmem, [anon:dalvik-DEX], memfd
    - bangcle  → anonymous mmap, /data/data private files
    - dpt      → anonymous, /data/app mmap
    - general  → any readable region with DEX magic
    Always skips: .jar, .oat, .vdex, .art, /system, /vendor, /apex
    """
    if not perm.startswith('r'):
        return False

    if not pathname:
        return True  # anonymous region — always scan

    p = pathname.lower()

    # ── Hard excludes ───────────────────────────────────────────────────
    for skip in ('/system/', '/vendor/', '/apex/', '/proc/', '/sys/',
                 '/dev/hw', '/dev/dri', '/dev/block',
                 '.oat', '.vdex', '.art', '.odex',
                 'boot.art', 'boot.oat',
                 '.jar'):           # system jars — never contain app DEX
        if skip in p:
            return False

    # ── Explicit includes ───────────────────────────────────────────────
    for inc in ('/dev/ashmem',     # ijiami decrypts into ashmem
                '/memfd:',         # memfd_create — used by some protectors
                'dalvik',          # [anon:dalvik-DEX] — ART runtime DEX
                '/data/app/',      # APK-mmap regions
                '/data/data/',     # app private data dir
                '/data/user/'):    # multi-user app data
        if inc in p:
            return True

    # ── app .so files (dpt loads real DEX via native lib) ──────────────
    if '.so' in p:
        if any(x in p for x in ('/system/', '/vendor/', '/apex/')):
            return False
        return True

    # ── anything else readable — scan it ───────────────────────────────
    return True


def scan_region(mem, start, end, perm, pathname):
    """Scan a memory region for DEX magic and save valid DEX files."""
    global mem_count
    region_size = end - start
    buf_offset  = 0
    leftover    = b''

    while buf_offset < region_size:
        read_size = min(CHUNK, region_size - buf_offset)
        try:
            mem.seek(start + buf_offset)
            chunk = mem.read(read_size)
        except Exception:
            break
        if not chunk:
            break

        data        = leftover + chunk
        search_from = 0

        while True:
            idx = data.find(DEX_MAGIC, search_from)
            if idx == -1:
                break

            abs_addr = start + buf_offset + idx - len(leftover)

            if len(data) < idx + 112:
                try:
                    mem.seek(abs_addr)
                    hdr = mem.read(112)
                except Exception:
                    search_from = idx + 1; continue
            else:
                hdr = data[idx:idx+112]

            if len(hdr) < 112:
                search_from = idx + 1; continue

            file_size  = struct.unpack_from('<I', hdr, 32)[0]
            endian_tag = struct.unpack_from('<I', hdr, 40)[0]

            if (file_size < MIN_SZ or file_size > MAX_SZ or
                    endian_tag != 0x12345678):
                search_from = idx + 1; continue

            sig = hdr[:64]
            if sig in seen_sigs:
                search_from = idx + 1; continue
            seen_sigs[sig] = True

            try:
                mem.seek(abs_addr)
                dex_data = mem.read(file_size)
            except Exception as e:
                print(f'[!] Read fail @ 0x{abs_addr:x}: {e}')
                search_from = idx + 1; continue

            if len(dex_data) < file_size:
                search_from = idx + 1; continue

            label    = pathname.rsplit('/', 1)[-1] if pathname else 'anon'
            out_path = os.path.join(OUT_DIR, f'mem_{mem_count}_{label}.dex')
            try:
                open(out_path, 'wb').write(dex_data)
                print(f'[+] #{mem_count}  {out_path}')
                print(f'    size={file_size:,}  addr=0x{abs_addr:x}  '
                      f'perm={perm}  src={pathname or "anon"}')
                mem_count += 1
            except Exception as e:
                print(f'[!] Write fail: {e}')

            search_from = idx + 1

        leftover   = data[-3:] if len(data) >= 3 else data
        buf_offset += read_size


def run_memory_scan(pass_num):
    print(f'\n[*] Memory scan pass {pass_num} (PID={PID}) ...')
    try:
        maps_data = open(maps_path).read()
        mem       = open(mem_path, 'rb')
    except Exception as e:
        print(f'[!] Cannot read /proc/{PID}: {e}')
        return

    regions = []
    for line in maps_data.splitlines():
        parts = line.split()
        if len(parts) < 2: continue
        perm     = parts[1]
        pathname = parts[5] if len(parts) >= 6 else ''
        if not should_scan(pathname, perm): continue
        try:
            s, e = [int(x, 16) for x in parts[0].split('-')]
        except ValueError:
            continue
        if (e - s) >= MIN_SZ:
            regions.append((s, e, perm, pathname))

    print(f'[*] Scanning {len(regions)} regions...')
    for s, e, perm, pn in regions:
        scan_region(mem, s, e, perm, pn)
    mem.close()
    print(f'[*] Pass {pass_num} done. Memory DEX found so far: {mem_count}')

# Auto-collect, rename & zip  (on-device, no PC needed)
# ═══════════════════════════════════════════════════════════════════════════

def collect_and_zip():
    """
    Priority: restored_* (dpt patched) > mem_* (live dump) > static_* (raw APK)
    Renames to classes_0.dex, classes_1.dex … and zips to /sdcard/dump_output.zip
    """
    import zipfile as _zf

    all_files = sorted(os.listdir(OUT_DIR))
    chosen    = []

    for f in all_files:
        if f.startswith('restored_') and f.endswith('.dex'):
            chosen.append(os.path.join(OUT_DIR, f))

    for f in all_files:
        if f.startswith('mem_') and f.endswith('.dex'):
            chosen.append(os.path.join(OUT_DIR, f))

    if not chosen:
        for f in all_files:
            if f.startswith('static_') and f.endswith('.dex'):
                chosen.append(os.path.join(OUT_DIR, f))

    if not chosen:
        print('[!] No DEX files to collect.')
        return

    col_dir = '/sdcard/dump_dex_efile'
    os.makedirs(col_dir, exist_ok=True)

    print(f'\n[*] ══════════════════════════════════════')
    print(f'[*] Collecting & renaming DEX files')
    print(f'[*] ══════════════════════════════════════')

    renamed = []
    for idx, src in enumerate(chosen):
        dst = os.path.join(col_dir, f'classes_{idx}.dex')
        with open(src, 'rb') as fi, open(dst, 'wb') as fo:
            fo.write(fi.read())
        sz = os.path.getsize(dst)
        print(f'  classes_{idx}.dex  ({sz:,} bytes)  ← {os.path.basename(src)}')
        renamed.append(dst)

    zip_path = '/sdcard/dump_output.zip'
    with _zf.ZipFile(zip_path, 'w', _zf.ZIP_STORED) as z:
        for f in renamed:
            z.write(f, os.path.basename(f))

    zsize = os.path.getsize(zip_path)
    print(f'\n[+] ZIP ready: {zip_path}  ({zsize:,} bytes, {len(renamed)} DEX)')
    print('    Open in file manager  OR  adb pull /sdcard/dump_output.zip ./')
