#!/system/bin/sh
#
# Offline APKProtector encN.dat -> DEX dumper for Termux.
#
# Simplest usage:
#   sh /storage/emulated/0/dump_apkprotector.sh
#
# With no argument, the script automatically processes:
#   /storage/emulated/0/Samsung.apk
#
# Optional custom input:
#   sh dump_apkprotector.sh /storage/emulated/0/another.apk
#
# Requirements in Termux:
#   pkg install openssl unzip
#   termux-setup-storage

set -u

DEFAULT_APK=/storage/emulated/0/Samsung.apk
if [ "$#" -eq 0 ] && [ -f "$DEFAULT_APK" ]; then
    ROOT=$DEFAULT_APK
else
    ROOT=${1:-/storage/emulated/0}
fi
OUT_DIR=${2:-/storage/emulated/0/apkprotector-dump}

# Termux normally provides $PREFIX/tmp. Do not assume /data/local/tmp exists
# or is writable, especially when the script is run without root.
if [ -n "${TMPDIR:-}" ] && [ -d "$TMPDIR" ]; then
    TMP_BASE=$TMPDIR
elif [ -n "${PREFIX:-}" ] && [ -d "$PREFIX/tmp" ]; then
    TMP_BASE=$PREFIX/tmp
else
    TMP_BASE=$OUT_DIR/.tmp
fi

die() {
    echo "Error: $*" >&2
    exit 1
}

need_command() {
    command -v "$1" >/dev/null 2>&1 ||
        die "missing '$1'. Install it in Termux first."
}

hex_bytes() {
    od -An -tx1 | tr -d ' \n'
}

safe_name() {
    printf '%s' "$1" | tr -c 'A-Za-z0-9._-' '_'
}

verify_dex() {
    dex=$1
    actual_size=$(wc -c < "$dex" | tr -d ' ')

    magic=$(dd if="$dex" bs=1 count=8 2>/dev/null | hex_bytes)
    case "$magic" in
        6465780a30*) ;;
        *)
            echo "  warning: output does not start with a DEX 03x magic ($magic)" >&2
            return 1
            ;;
    esac

    declared_size=$(od -An -tu4 -j 32 -N 4 "$dex" 2>/dev/null | tr -d ' ')
    if [ -n "$declared_size" ] && [ "$declared_size" -ne "$actual_size" ]; then
        echo "  warning: DEX size mismatch (declared $declared_size, actual $actual_size)" >&2
        return 1
    fi

    stored_sha1=$(
        dd if="$dex" bs=1 skip=12 count=20 2>/dev/null |
            hex_bytes
    )
    calculated_sha1=$(
        dd if="$dex" bs=4096 skip=0 2>/dev/null |
            tail -c +33 |
            openssl dgst -sha1 -binary 2>/dev/null |
            hex_bytes
    )

    if [ "$stored_sha1" != "$calculated_sha1" ]; then
        echo "  warning: DEX SHA-1 mismatch" >&2
        echo "    stored:    $stored_sha1" >&2
        echo "    calculated: $calculated_sha1" >&2
        return 1
    fi

    echo "  verified: DEX magic, declared size, and SHA-1 match"
    return 0
}

decrypt_enc() {
    input=$1
    label=$2
    size=$(wc -c < "$input" | tr -d ' ')

    if [ "$size" -lt 88 ]; then
        echo "Skipping $label: file is too short ($size bytes)" >&2
        return 1
    fi

    body_size=$((size - 72))
    if [ $((body_size % 16)) -ne 0 ]; then
        echo "Skipping $label: encrypted body is not 16-byte aligned" >&2
        return 1
    fi

    magic=$(dd if="$input" bs=1 count=8 2>/dev/null | hex_bytes)
    case "$magic" in
        6465780a30*) ;;
        *)
            echo "Skipping $label: unexpected clear header ($magic)" >&2
            return 1
            ;;
    esac

    # APKProtector rule recovered from the native loader:
    #   key = SHA-256(bytes 8..55)[:16]
    #   iv  = bytes 56..71
    key=$(
        dd if="$input" bs=1 skip=8 count=48 2>/dev/null |
            openssl dgst -sha256 -binary 2>/dev/null |
            hex_bytes |
            cut -c1-32
    )
    iv=$(
        dd if="$input" bs=1 skip=56 count=16 2>/dev/null |
            hex_bytes
    )

    [ "${#key}" -eq 32 ] || die "could not derive a 16-byte AES key"
    [ "${#iv}" -eq 32 ] || die "could not read a 16-byte IV"

    tmp_dir=$(mktemp -d "$TMP_BASE/apkprotector.XXXXXX") ||
        die "could not create a temporary directory"
    plain="$tmp_dir/plain.dex"
    output="$OUT_DIR/$(safe_name "$label").recovered.dex"

    # Skip the 72-byte container header and let OpenSSL remove PKCS#7 padding.
    if ! dd if="$input" bs=72 skip=1 2>/dev/null |
        openssl enc -d -aes-128-cbc -K "$key" -iv "$iv" -out "$plain" 2>"$tmp_dir/openssl.err"
    then
        echo "Failed to decrypt $label:" >&2
        cat "$tmp_dir/openssl.err" >&2
        rm -rf "$tmp_dir"
        return 1
    fi

    # The decrypted stream has the original DEX contents, but its first eight
    # bytes are replaced by the clear magic stored in the container.
    dd if="$input" bs=1 count=8 2>/dev/null > "$output"
    dd if="$plain" bs=1 skip=8 2>/dev/null >> "$output"

    echo "Dumped: $output"
    echo "  key: $key"
    echo "  iv:  $iv"
    if ! verify_dex "$output"; then
        echo "  output was retained for inspection despite verification failure" >&2
    fi

    rm -rf "$tmp_dir"
    return 0
}

process_apk() {
    apk=$1
    apk_name=$(basename "$apk")
    apk_stem=${apk_name%.apk}
    list_file=$(mktemp "$TMP_BASE/apkprotector-list.XXXXXX") ||
        die "could not create temporary file"

    if ! unzip -Z1 "$apk" 2>/dev/null |
        grep '^assets/enc[0-9][0-9]*\.dat$' > "$list_file"
    then
        rm -f "$list_file"
        echo "No assets/encN.dat payloads found in $apk"
        return 1
    fi

    found=0
    while IFS= read -r entry; do
        [ -n "$entry" ] || continue
        found=1
        asset_name=$(basename "$entry")
        temp_asset=$(mktemp "$TMP_BASE/apkprotector-asset.XXXXXX") ||
            die "could not create temporary asset"
        unzip -p "$apk" "$entry" > "$temp_asset" || {
            echo "Could not extract $entry from $apk" >&2
            rm -f "$temp_asset"
            continue
        }
        decrypt_enc "$temp_asset" "${apk_stem}.${asset_name}" || true
        rm -f "$temp_asset"
    done < "$list_file"

    rm -f "$list_file"
    [ "$found" -eq 1 ]
}

process_path() {
    path=$1
    if [ -f "$path" ]; then
        case "$path" in
            *.apk|*.APK) process_apk "$path" ;;
            *enc[0-9]*.dat) decrypt_enc "$path" "$(basename "$path" .dat)" ;;
            *)
                echo "Skipping unsupported file: $path" >&2
                return 1
                ;;
        esac
    elif [ -d "$path" ]; then
        find "$path" -type f \( -iname '*.apk' -o -name 'enc[0-9]*.dat' \) |
            while IFS= read -r item; do
                process_path "$item" || true
            done
    else
        die "path does not exist: $path"
    fi
}

need_command dd
need_command od
need_command openssl
need_command mktemp
need_command unzip
need_command find
need_command grep
need_command tail
need_command tr
need_command cut
need_command wc

mkdir -p "$OUT_DIR" "$TMP_BASE" || die "could not create output or temporary directory"
process_path "$ROOT"
